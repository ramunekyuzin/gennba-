# PDF rename (store_amount) + sort by month + compressed copy. PowerShell + Word. ASCII-only source. param([string]$Target = $PSScriptRoot) $MOVE_TO_MONTH = $true # $false = rename only, no folder move $COMPRESS = $true # $true = also save a smaller copy into "keiryou" subfolder (original is kept) $COMPRESS_MIN_KB = 100 # skip compression for files smaller than this (KB) $Target = (Resolve-Path $Target).Path Write-Host "Target: $Target`n" $NEN = [string][char]0x5E74 $GATSU = [string][char]0x6708 $EN = [string][char]0x5186 $MIBUNRUI = -join ([char]0x672A, [char]0x5206, [char]0x985E) $KEIRYOU = -join ([char]0x8EFD, [char]0x91CF, [char]0x7248) $pdfs = Get-ChildItem -Path $Target -Filter *.pdf -File if ($pdfs.Count -eq 0) { Write-Host "No PDF found." Read-Host "Press Enter to close" exit } $patterns = @( @{ re = '(20\d{2})\s*\u5E74\s*(\d{1,2})\s*\u6708'; kind = 'western' }, @{ re = '(20\d{2})[/\-\.\uFF0E\uFF0F](\d{1,2})[/\-\.\uFF0E\uFF0F]\d{1,2}'; kind = 'western' }, @{ re = '\u4EE4\u548C\s*(\d{1,2})\s*\u5E74\s*(\d{1,2})\s*\u6708'; kind = 'reiwa' }, @{ re = '[R\uFF32]\s*(\d{1,2})[\.\uFF0E\u5E74/]\s*(\d{1,2})[\.\uFF0E\u6708/]'; kind = 'reiwa' } ) function Find-Month([string]$text) { foreach ($p in $patterns) { $m = [regex]::Match($text, $p.re) if ($m.Success) { $y = [int]$m.Groups[1].Value $mo = [int]$m.Groups[2].Value if ($p.kind -eq 'reiwa') { $y += 2018 } if ($mo -ge 1 -and $mo -le 12) { return ('{0}{1}{2:00}{3}' -f $y, $NEN, $mo, $GATSU) } } } return $null } function Normalize-Num([string]$s) { $s = $s -replace '[\uFF0C,]', '' $sb = '' foreach ($c in $s.ToCharArray()) { $code = [int]$c if ($code -ge 0xFF10 -and $code -le 0xFF19) { $sb += [char]($code - 0xFF10 + 48) } else { $sb += $c } } $out = 0 if ([long]::TryParse($sb, [ref]$out)) { return $out } return 0 } # Amount: prefer number near total keywords, else largest yen-marked number function Find-Amount([string]$text) { $num = '[\u00A5\uFFE5]?\s*([0-9\uFF10-\uFF19][0-9,\uFF10-\uFF19\uFF0C]*)\s*(\u5186|-)?' $keys = '(\u5408\u8A08|\u7A0E\u8FBC|\u304A\u8CB7\u4E0A|\u8A08|\u5408\u8A08\u91D1\u984D|\u5FA1\u5408\u8A08)' $best = $null $m = [regex]::Matches($text, $keys + '[^0-9\u00A5\uFFE5\uFF10-\uFF19]{0,15}' + $num) foreach ($x in $m) { $v = Normalize-Num $x.Groups[2].Value if ($v -gt 0 -and ($best -eq $null -or $v -gt $best)) { $best = $v } } if ($best -ne $null) { return $best } $m = [regex]::Matches($text, '[\u00A5\uFFE5]\s*([0-9\uFF10-\uFF19][0-9,\uFF10-\uFF19\uFF0C]*)') foreach ($x in $m) { $v = Normalize-Num $x.Groups[1].Value if ($v -gt 0 -and ($best -eq $null -or $v -gt $best)) { $best = $v } } return $best } # Store name: first meaningful line (skip receipt title / date / customer lines) function Find-Store([string]$text) { $skip = '(\u9818\u53CE|\u30EC\u30B7\u30FC\u30C8|\u6A23|\u65E5\u4ED8|\u767A\u884C|20\d{2}|\u4EE4\u548C|^[\d\s/\-:]+$|\u5408\u8A08|\u91D1\u984D)' foreach ($line in ($text -split "[\r\n\a\f]+")) { $l = $line.Trim() if ($l.Length -lt 2) { continue } if ($l -match $skip) { continue } $l = $l -replace '[\\/:*?"<>|]', '' if ($l.Length -gt 30) { $l = $l.Substring(0, 30) } return $l } return $null } function Unique-Path([string]$dir, [string]$base, [string]$ext, [string]$self) { $p = Join-Path $dir ($base + $ext) $n = 1 while ((Test-Path $p) -and ($p -ne $self)) { $p = Join-Path $dir ("{0}_{1}{2}" -f $base, $n, $ext) $n++ } return $p } $word = New-Object -ComObject Word.Application $word.Visible = $false $word.DisplayAlerts = 0 $stats = @{ ok = 0; small = 0; skip = 0 } foreach ($pdf in $pdfs) { $dest = $MIBUNRUI $newBase = $pdf.BaseName $doc = $null $tmpOut = $null try { $doc = $word.Documents.Open($pdf.FullName, $false, $true, $false) $text = $doc.Content.Text if ([string]::IsNullOrWhiteSpace($text)) { $dest = $MIBUNRUI + "(no text)" } else { $head = $text.Substring(0, [Math]::Min(3000, $text.Length)) $found = Find-Month $head if ($found) { $dest = $found } $store = Find-Store $head $amount = Find-Amount $head if ($store -and $amount) { $newBase = '{0}_{1}{2}' -f $store, $amount, $EN } elseif ($store) { $newBase = $store } elseif ($amount) { $newBase = '{0}{1}' -f $amount, $EN } } # Compressed copy: export to temp with "minimum size" option (17 = PDF, 1 = wdExportOptimizeForMinimumSize) if ($COMPRESS -and ($pdf.Length -ge $COMPRESS_MIN_KB * 1024)) { $tmpOut = Join-Path $env:TEMP ("pdfsort_" + [guid]::NewGuid().ToString("N") + ".pdf") $doc.ExportAsFixedFormat($tmpOut, 17, $false, 1, 0, 1, 1, 0, $false, $false, 0, $false, $true, $false) } $doc.Close(0) $doc = $null } catch { Write-Host "x $($pdf.Name): failed ($($_.Exception.Message))" $dest = $MIBUNRUI + "(error)" if ($doc -ne $null) { try { $doc.Close(0) } catch {} ; $doc = $null } } $destDir = $Target if ($MOVE_TO_MONTH) { $destDir = Join-Path $Target $dest if (-not (Test-Path $destDir)) { New-Item -ItemType Directory -Path $destDir | Out-Null } } $destPath = Unique-Path $destDir $newBase $pdf.Extension $pdf.FullName $origSize = $pdf.Length if ($destPath -ne $pdf.FullName) { Move-Item -Path $pdf.FullName -Destination $destPath } Write-Host "o $($pdf.Name) -> $destPath" if ($tmpOut -and (Test-Path $tmpOut)) { $newSize = (Get-Item $tmpOut).Length if ($newSize -lt $origSize * 0.9) { $kDir = Join-Path $destDir $KEIRYOU if (-not (Test-Path $kDir)) { New-Item -ItemType Directory -Path $kDir | Out-Null } $kBase = [System.IO.Path]::GetFileNameWithoutExtension($destPath) $kPath = Unique-Path $kDir $kBase ".pdf" "" Move-Item -Path $tmpOut -Destination $kPath Write-Host (" light: {0:N0} KB -> {1:N0} KB ({2})" -f ($origSize/1KB), ($newSize/1KB), $kPath) $stats.ok++ } else { Remove-Item $tmpOut -ErrorAction SilentlyContinue Write-Host (" light: no gain ({0:N0} KB -> {1:N0} KB), skipped" -f ($origSize/1KB), ($newSize/1KB)) $stats.small++ } } elseif ($COMPRESS) { $stats.skip++ } } $word.Quit() [System.Runtime.InteropServices.Marshal]::ReleaseComObject($word) | Out-Null if ($COMPRESS) { Write-Host ("`nLight copies: {0} saved, {1} no gain, {2} skipped (small/error)" -f $stats.ok, $stats.small, $stats.skip) } Write-Host "`nDone" Read-Host "Press Enter to close"
